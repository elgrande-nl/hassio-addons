#include <tunables/global>

  # Allow the container runtime to stop the Python process normally.
  signal (receive),

  /usr/local/bin/python rix,
  /usr/local/bin/python3 rix,
  /usr/local/bin/python3.13 rix,
  /usr/local/lib/ r,
  /usr/local/lib/** mr,
  /usr/lib/ r,
  /usr/lib/** mr,
  /lib/ r,
  /lib/** mr,

  # Application code, dashboard, options and certificate/SSH configuration.
  /app/ r,
  /app/** r,
  /data/ r,
  /data/** r,
  /etc/ld-musl-*.path r,
  /etc/hosts r,
  /etc/hostname r,
  /etc/resolv.conf r,
  /etc/nsswitch.conf r,
  /etc/gai.conf r,
  /etc/passwd r,
  /etc/group r,
  /etc/localtime r,
  /etc/ssl/ r,
  /etc/ssl/** r,
  /etc/ssh/ r,
  /etc/ssh/** r,
  /usr/share/zoneinfo/** r,
  /sys/devices/system/cpu/online r,

  /share/ r,
  /share/** r,
  /share/birdbuddy/ rw,
  /share/birdbuddy/** rwk,
  /tmp/ rw,

}
